cmake_minimum_required(VERSION 3.16)
project(wonik_gazebo)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(qd_kinematics REQUIRED)
find_package(gazebo_dev REQUIRED)
find_package(gazebo REQUIRED)
add_library(qd_kinematic_base SHARED src/qd_kinematic_base_plugin.cpp)
target_include_directories(qd_kinematic_base PRIVATE ${GAZEBO_INCLUDE_DIRS})
ament_target_dependencies(qd_kinematic_base qd_kinematics)
target_link_libraries(qd_kinematic_base ${GAZEBO_LIBRARIES})
install(TARGETS qd_kinematic_base LIBRARY DESTINATION lib)
install(PROGRAMS scripts/fault_injection.py scripts/camera_relay.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch worlds config DESTINATION share/${PROJECT_NAME})
ament_package()
